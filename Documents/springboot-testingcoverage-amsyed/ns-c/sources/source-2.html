 /**

     )
     @ApiResponses(value = {
             @ApiResponse(
                     responseCode = "201",

                     content = @Content(mediaType = "application/json",
                             schema = @Schema(implementation = EventResponseDTO.class))

             @ApiResponse(
                     responseCode = "400",
                     description = "Invalid request body (validation error or bad data)",
                     content = @Content

                     example = "5"
